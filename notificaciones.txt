            user_id,
            gym_id,
            id_agente_emisor,
            tipo,
            prioridad,
            titulo,
            mensaje,
            payload_json,
            action_url,
            leido,
            id_miembro_destinatario
        ) VALUES (
            :user_id,
            :gym_id,
            :id_agente_emisor,
            :tipo,
            :prioridad,
            :titulo,
            :mensaje,
            :payload_json,
            :action_url,
            :leido,
            :id_miembro_destinatario
        )
    ";

    $stmt = $conn->prepare($sql);

    $stmt->execute([
        ':user_id' => $user_id,
        ':gym_id' => $gym_id,
        ':id_agente_emisor' => $id_agente_emisor,
        ':tipo' => $tipo,

    ]);
